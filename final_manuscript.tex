\begin{abstract}
We present a rigorous technical assessment of the weak-field phenomenology arising from modified gravity formulations in the AQUAL family. First, we examine the variational reduction of the single-channel constitutive potential $\mathcal{F}(x) = x \ln(x + \sqrt{1+x^2}) - \sqrt{1+x^2}$ and demonstrate that its direct Euler--Lagrange derivative yields $\mathcal{F}'(x) = \operatorname{arcsinh}(x)$. The transition to the standard rational interpolation function $\mu_{\text{simple}}(x) = x/\sqrt{1+x^2}$ is shown to constitute an auxiliary constitutive boundary projection hypothesis $\mathbf{[O]}$, rather than an unconditioned consequence of the Euler--Lagrange action variation. Second, we analyze the horizon acceleration scale $a_0 = cH_0/(2\pi)$, showing through dimensionally consistent horizon thermodynamics ($T_{\text{GH}} = \hbar H_0 / (2\pi k_B)$ and $T_U = \hbar a / (2\pi c k_B)$) that temperature matching yields $a = cH_0$ where the $2\pi$ KMS factors cancel; the additional $1/(2\pi)$ divisor is thus treated explicitly as an open boundary normalization $\mathbf{[O]}$. Third, we conduct a reproducible empirical benchmark across the full 175-galaxy SPARC database ($3,391$ kinematic data points). The strict zero-parameter formulation yields median $\chi^2_{\text{data}}/N_g = 29.12$ and aggregate $\sum\chi^2_{\text{data}}/\text{dof} = 144.04$, outperforming fixed baryons-only baselines ($\chi^2/\text{dof} = 204.65$ for unit $M/L$, $406.49$ for standard SPARC) yet remaining a poor absolute fit. In contrast, an in-sample Maximum A Posteriori (MAP) nuisance optimization ($N_{\text{par}}=382$) yields median $\chi^2_{\text{data}}/N_g = 2.88$ and nominal aggregate $\sum\chi^2_{\text{data}}/\text{dof}_{\text{nom}} = 7.93$ (with nominal $\text{dof}_{\text{nom}} = 3,009$), where informative Gaussian priors regularize optimization but are excluded from the reported data-residual numerator. Finally, we document the formal verification boundary: Lean 4 mechanically checks the mathematical soundness of encoded theorems $\mathbf{[P]}$ under axiomatic foundationals, but does not certify physical theory selection. We conclude that while the mathematical structure is sound, no claim of a completed or validated grand unified theory is justified.
\end{abstract}

\section{Effective Weak-Field Model}
\label{sec:model}

In the weak-field, non-relativistic limit, the modified gravitational potential $\Phi$ satisfies the non-linear Poisson equation \cite{Bekenstein1984}:
\begin{equation}
\label{eq:aqual_field}
\nabla \cdot \left[ \mu\left(\frac{|\nabla\Phi|}{a_0}\right) \nabla\Phi \right] = 4\pi G \rho_{\text{bar}},
\end{equation}
where $\rho_{\text{bar}}$ is the baryonic mass density, $a_0$ is the characteristic acceleration scale, and $\mu(x)$ is an interpolation function satisfying $\mu(x) \to 1$ for $x \gg 1$ and $\mu(x) \to x$ for $x \ll 1$.

Under spherical or planar symmetry, Equation (\ref{eq:aqual_field}) reduces to the algebraic relation between the total gravitational acceleration $g = |\nabla\Phi|$ and the Newtonian baryonic acceleration $g_{\text{bar}} = |\nabla\Phi_N|$:
\begin{equation}
\label{eq:mu_relation}
g \cdot \mu\left(\frac{g}{a_0}\right) = g_{\text{bar}}.
\end{equation}
For the "simple" interpolation function $\mu_{\text{simple}}(x) = \frac{x}{\sqrt{1+x^2}}$ \cite{FamaeyBinney2005}, Equation (\ref{eq:mu_relation}) yields the closed-form physical acceleration:
\begin{equation}
\label{eq:v_predict}
g = g_{\text{bar}} \left[ \frac{1}{2} + \sqrt{\frac{1}{4} + \frac{a_0}{g_{\text{bar}}}} \right].
\end{equation}

\section{Exact Action Variation and the Open Constitutive Closure}
\label{sec:action_variation}

The scalar sector of the Lagrangian is described by the single-channel AQUAL action \cite{Bekenstein1984}:
\begin{equation}
S_{\text{AQUAL}} = \int d^4x \left[ -\frac{1}{8\pi G} \nabla\Phi_N \cdot \nabla\Phi - \frac{a_0^2}{4\pi G} \mathcal{F}\left( \frac{|\nabla\Phi|}{a_0} \right) \right], \qquad x \equiv \frac{|\nabla\Phi|}{a_0}.
\end{equation}
The proposed constitutive potential is:
\begin{equation}
\mathcal{F}(x) = x \ln\left( x + \sqrt{1+x^2} \right) - \sqrt{1+x^2} = x \operatorname{arcsinh}(x) - \sqrt{1+x^2}.
\end{equation}

\subsection{Direct Variational Derivative}
Differentiating $\mathcal{F}(x)$ with respect to $x$ yields:
\begin{equation}
\mathcal{F}'(x) = \operatorname{arcsinh}(x) + \frac{x}{\sqrt{1+x^2}} - \frac{x}{\sqrt{1+x^2}} = \operatorname{arcsinh}(x).
\end{equation}
The direct Euler--Lagrange field equation resulting from this action is:
\begin{equation}
\nabla \cdot \left[ \operatorname{arcsinh}\left(\frac{|\nabla\Phi|}{a_0}\right) \frac{\nabla\Phi}{|\nabla\Phi|} \right] = \frac{4\pi G}{a_0} \rho_{\text{bar}}.
\end{equation}

\subsection{The Epistemic Boundary $\mathbf{[O]}$}
The transition from $\mathcal{F}'(x) = \operatorname{arcsinh}(x)$ to the rational function $\mu_{\text{simple}}(x) = \frac{x}{\sqrt{1+x^2}}$ does not follow from variational calculus alone. The assertion that planar disk surface balance projects the derivative onto the direct gradient ratio is an \textbf{auxiliary constitutive closure hypothesis $\mathbf{[O]}$}. While the mathematical properties of $\mu_{\text{simple}}(x)$ are verified $\mathbf{[P]}$ in Lean 4 (\texttt{MuProjection.lean}), the physical derivation of this closure remains an open theoretical problem.

\section{Acceleration-Scale Normalization: Dimensional Analysis and Open Boundary}
\label{sec:a0_norm}

The framework adopts the cosmological horizon acceleration relation:
\begin{equation}
a_0 = \frac{c H_0}{2\pi} \approx 1.042 \times 10^{-10}\text{ m/s}^2 \quad (H_0 = 67.4\text{ km/s/Mpc}) \quad \mathbf{[O]}.
\end{equation}

\subsection{Thermal Horizon Dimensional Analysis}
Consider the Gibbons--Hawking temperature of a de Sitter cosmological horizon with Hubble parameter $H_0$ \cite{GibbonsHawking1977}:
\begin{equation}
T_{\text{GH}} = \frac{\hbar H_0}{2\pi k_B} \quad ([H_0] = \text{s}^{-1}),
\end{equation}
and the Unruh temperature associated with an observer experiencing constant acceleration $a$ \cite{Unruh1976}:
\begin{equation}
T_U = \frac{\hbar a}{2\pi c k_B}.
\end{equation}
Equating horizon thermal states ($T_U = T_{\text{GH}}$) yields:
\begin{equation}
\frac{\hbar a}{2\pi c k_B} = \frac{\hbar H_0}{2\pi k_B} \implies a = c H_0.
\end{equation}
The universal KMS factor $2\pi$ cancels identically. Therefore, thermal equilibrium derives $a = cH_0$, not $a_0 = \frac{cH_0}{2\pi}$. The additional $1/(2\pi)$ divisor is an \textbf{open boundary normalization $\mathbf{[O]}$}, rather than a derived first-principles result.
